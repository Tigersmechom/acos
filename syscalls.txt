man syscalls - список сисколов

starce ./prog - выведет список всех вызванных сисколов

дескриптор - уникальный номер (0 - cin, 1 - cout, 2 - cerr (это все в терминале))

write(int desc, void* ptr, size_t bytes) - можео поставить desc = 21 и в терминале: 21>out.txt, 
и возвращает число: сколько байт записано

read(int desc, void* ptr, size_t bytes) - прочитать откуда куда и сколько, и возвращает то, сколько прочитал байт
(у файла указатель offset двигается вперед невозвратно (далее read начнется оттуда где этот остановился))

open("path", flags, mask) возвращает файловый дескриптор (fd)
mask = 0644 - типо если создаю новый файл, то какие права у него будут(rw-r--r--)
flags:
O_RDONLY
O_WRONLY (перезаписывает с самого начала)
O_RDWR
O_CREATE - создать если нет
O_TRUNC - обнулить
O_APPEND - писать в конец

chmod("path", mask)
fchmod(fd, mask) - эти сисколы меняют права доступа

struct stat st;
fstat(fd, &st); - syscall, который узнает много данных о файле (st.st_size - размер файла, если неизвестно)

void* mmap( (!!!все подгрузки physical RAM ленивые по мере обращения (ну или к ssd)) (все работы на page_size!!!)
  void* addr
  size_t len
  int prot
  int flags
  int fd
  off_t offset
)
-addr: если nullptr, то сам ищет адрес, иначе ... (но можно указать флаг MAP_FIXED_NOREPLACE и тогда если 
эта память занята, то mmap вернет ошибку EEXIST (mmap_return == MAP_FAILED))
(должен быть выровнен по page_size) (это просто hint, что хочу тут, но далеко не всегда получится (может выбрать другое место))
-len: сколько байт
-prot: PROT_READ, PROT_WRITE, PROT_EXEC(обычно делают флаги write/read, а потом exec/read в целях безопасности 
(через mprotect) и потом можно вызывать функции по этим указателям (то есть можно в runtime дописывать asm код)), 
PROT_NONE (это обычно вокруг стека (чтобы получить ошибку stack overflow))
-flags: MAP_ANONYMUS(зануляет физическую память (все равно physical pages лениво подгружаются)), 
MAP_SHARED(несколько процессов могут записывать/читать данные
из одной и той же физической памяти (запись происходит сразу на RAM (не лениво через cache (!!!гонка за данными 
все равно может быть)))), 
MAP_PRIVATE(изменения не применяютс к RAM и ssd даже после munmap(private страница просто исчезнет) 
(то есть создается копия локальная и другие потоки не видят изменения)) (msync все равно не изменит данные на ssd),
MAP_FIXED_NOREPLACE (безопасно хочу свой адрес получить)
MAP_FIXED может затереть данные, если по этому адресу что-то было уже (это приказ, выделить вот тут память)
MAP_HUGETLB и MAP_HUGE_1GB (то есть страницы не по 4кб, а 1 огромная страница (в RAM тоже память подряд идет))
(!!!offset должен быть кратен размеру страницы(если offset нужен))
(если компьютер работает долго, то RAM может быть неудачно фрагментировано и hugetable не получится)
(!!!huge_tables не swappable (на ssd не переедут если ram закончится))
MAP_POPULATE - говорит, что подгрузи все physical pages сразу, чтобы в runtime было меньще page fault
(однако если будут проблемы, то page fault в runtime все равно встретятся
(linux говорит, что из-за этого если что я не хочу ломаться, если сразу не получается все выделить))
-fd: если -1, то просто возвращает память, если написать fd(через open), то прочитает и запишет len байт
-offset: говорит, что начни читать файл вот с этого индекса (!!!offset должен бть кратен размеру таблицы
(поэтому там надо аккуратно, и лучше прочитать сильно больше))

munmap(ptr, bytes_len) - вернуть эту память взад (ptr должен быть выровнен по page_size, а len - необязательно
(однако все равно удалится вся страница (типо len округляется до ближайшего размера страницы (2000 -> 4096))))
(до munmap нет гарантии что msync произойдет)
(можно несколько раз одну и ту же память освобождать(по крайней мере на linux), но нежелательно)

msync(ptr, size, flag) - помогает применить изменения и записать сразу на ssd (flag - MS_SYNC, MS_INVALIDATE - просит 
инвалидировать все другие cached mappings соответствующих данных)
!!! ptr должен быть выровнен по размеру страницы

mprotect(ptr, bytes_len, flags) flags = PROT_EXEC | PROT_READ |... - изменяет разрешения этой страницы
